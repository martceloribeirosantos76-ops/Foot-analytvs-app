class HomePage extends StatelessWidget {
  const HomePage({super.key});

  @override

          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          const Text(
            'Análises de Futebol',
            style: TextStyle(
              fontSize: 28,
              fontWeight: FontWeight.bold,
            ),
          ),
          const SizedBox(height: 6),
          Text(
            'Dados, estatísticas e mercados em um só lugar.',
            style: TextStyle(color: Colors.grey.shade400),
          ),
          const SizedBox(height: 20),

          _infoCard(
            icon: Icons.cloud_off,
            title: 'API ainda não conectada',
            text:
                'As probabilidades e estatísticas reais aparecerão quando a fonte de dados for conectada.',
            color: Colors.orange,
          ),

          const SizedBox(height: 16),

          _sectionTitle('PRINCIPAIS ÁREAS'),

          _menuCard(
            context,
            icon: Icons.sports_soccer,
            title: 'Jogos de hoje',

            title: 'Times',

            page: const TeamsPage(),
          ),

          _menuCard(
            context,
            icon: Icons.person,
            title: 'Jogadores',
            subtitle: 'Estatísticas e comparação individual.',
            page: const PlayersPage(),
          ),

          const SizedBox(height: 18),
          _sectionTitle('MERCADOS DISPONÍVEIS'),

          _marketGrid(context),
        ],
      ),
    );
  }

  Widget _marketGrid(BuildContext context) {
    final markets = [
      ['1X2', Icons.sports_soccer],
      ['Dupla chance', Icons.sync_alt],
      ['Empate anula', Icons.shield],
      ['Over / Under', Icons.bar_chart],
      ['Ambos marcam', Icons.groups],
      ['Escanteios', Icons.flag],
      ['Cartões', Icons.style],
      ['Finalizações', Icons.gps_fixed],
      ['Chutes no gol', Icons.adjust],
      ['Faltas', Icons.pan_tool],
      ['Impedimentos', Icons.flag_outlined],
      ['1º tempo', Icons.timer],
      ['Handicap', Icons.exposure_plus_1],
      ['Primeiro a marcar', Icons.emoji_events],
    ];

    return GridView.builder(
      shrinkWrap: true,
      physics: const NeverScrollableScrollPhysics(),
      itemCount: markets.length,
      gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
        crossAxisCount: 2,
        crossAxisSpacing: 10,
        mainAxisSpacing: 10,
        childAspectRatio: 1.65,
      ),
      itemBuilder: (context, index) {
        return Card(
          child: Padding(
            padding: const EdgeInsets.all(12),
            child: Row(
              children: [
                Icon(
                  markets[index][1] as IconData,
                  color: Colors.greenAccent,
                ),
                const SizedBox(width: 10),
                Expanded(
                  child: Text(
                    markets[index][0] as String,
                    style: const TextStyle(
                      fontWeight: FontWeight.w600,
                    ),
                  ),
                ),
              ],
            ),
          ),
        );
      },
    );
  }

  Widget _sectionTitle(String text) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 10),
      child: Text(
        text,
        style: TextStyle(
          color: Colors.greenAccent.shade400,
          fontWeight: FontWeight.bold,
          letterSpacing: 1.2,
        ),
      ),
    );
  }

  Widget _infoCard({
    required IconData icon,
    required String title,
    required String text,
    required Color color,
  }) {
    return Card(
      color: color.withOpacity(.10),
      child: Padding(
        padding: const EdgeInsets.all(16),
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Icon(icon, color: color),
            const SizedBox(width: 12),
            Expanded(
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text(
                    title,
                    style: TextStyle(
                      color: color,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const SizedBox(height: 5),
                  Text(
                    text,
                    style: TextStyle(color: Colors.grey.shade300),
                  ),
                ],
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _menuCard(
    BuildContext context, {
    required IconData icon,
    required String title,
    required String subtitle,
    required Widget page,
  }) {
    return Card(
      child: ListTile(
        leading: CircleAvatar(
          backgroundColor: Colors.green.withOpacity(.15),
          child: Icon(icon, color: Colors.greenAccent),
        ),
        title: Text(
          title,
          style: const TextStyle(fontWeight: FontWeight.bold),
        ),
        subtitle: Text(subtitle),
        trailing: const Icon(Icons.chevron_right),
        onTap: () {
          Navigator.push(
            context,
            MaterialPageRoute(builder: (_) => page),
          );
        },
      ),
    );
  }
}

class MatchesPage extends StatelessWidget {
  const MatchesPage({super.key});

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Jogos de hoje')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          _statusCard(),

            context,

            'Time visitante',

          _matchCard(

          ),
        ],
      ),
    );
  }

  Widget _statusCard() {
    return Card(
      child: Padding(
        padding: const EdgeInsets.all(16),
        child: Row(
          children: [
            const Icon(Icons.cloud_off, color: Colors.orange),
            const SizedBox(width: 12),
            Expanded(
              child: Text(
                'Nenhum jogo real carregado. Conecte a API para obter partidas.',
                style: TextStyle(color: Colors.grey.shade300),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _matchCard(
    BuildContext context,
    String home,
    String away,
  ) {
    return Card(
      child: ListTile(
        leading: const Icon(
          Icons.sports_soccer,
          color: Colors.greenAccent,
        ),
        title: Text('$home  x  $away'),
        subtitle: const Text('Análise disponível após conexão da API'),
        trailing: const Icon(Icons.chevron_right),
        onTap: () {
          Navigator.push(
            context,
            MaterialPageRoute(
              builder: (_) => MatchAnalysisPage(
                home: home,
                away: away,
              ),
            ),
          );
        },
      ),
    );
  }
}

class MarketSection extends StatelessWidget {
  final String title;
  final List<String> markets;

  const MarketSection({
    super.key,
    required this.title,
    required this.markets,
  });

  @override
  Widget build(BuildContext context) {
    return Card(
      margin: const EdgeInsets.only(bottom: 14),
      child: ExpansionTile(
        title: Text(
          title,
          style: const TextStyle(fontWeight: FontWeight.bold),
        ),
        children: markets.map((market) {
          return ListTile(
            leading: const Icon(
              Icons.analytics_outlined,
              color: Colors.greenAccent,
            ),
            title: Text(market),
            trailing: const Text(
              'API',
              style: TextStyle(
                color: Colors.orange,
                fontWeight: FontWeight.bold,
              ),
            ),
          );
        }).toList(),
      ),
    );
  }
}
